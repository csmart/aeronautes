import { afterEach, expect, it, vi } from "vitest";
import { flushSync, mount, unmount } from "svelte";
import type { AppSettingsDTO } from "../../lib/api";
import { BrowserState } from "../browser/browserState.svelte";
import { newFakeApi } from "../browser/fakeApi";
import JobsDrawer from "./JobsDrawer.svelte";

const initial: AppSettingsDTO = {
  maxActiveJobs: 2,
  defaultProfile: "balanced",
  auditRetentionDays: 17,
  jobRetentionDays: 11,
};
let drawer: ReturnType<typeof mount> | null = null;
const pendingCleanup: (() => void)[] = [];

afterEach(async () => {
  for (const release of pendingCleanup.splice(0)) release();
  await Promise.resolve();
  if (drawer) await unmount(drawer);
  drawer = null;
  document.body.innerHTML = "";
  vi.useRealTimers();
  console.log("BUDGET_PROBE_CLEANUP", JSON.stringify({
    drawerUnmounted: true, pendingFallbacksReleased: true, bodyEmpty: document.body.innerHTML === "",
  }));
});

function deferred<T>(fallback: T) {
  let resolve!: (value: T) => void;
  let reject!: (reason: Error) => void;
  const promise = new Promise<T>((done, fail) => {
    resolve = done;
    reject = fail;
  });
  pendingCleanup.push(() => resolve(fallback));
  return { promise, resolve, reject };
}

function fixture() {
  const api = newFakeApi();
  api.jobs = vi.fn(async () => []);
  api.appSettings = vi.fn(async () => ({ ...initial }));
  api.saveAppSettings = vi.fn(async (settings) => ({ ...settings }));
  const browser = new BrowserState(api);
  return { api, browser };
}

function budgetSelect(): HTMLSelectElement | null {
  const label = [...document.querySelectorAll("label")].find((element) =>
    element.querySelector("span")?.textContent?.trim() === "Active-job budget");
  return label?.querySelector("select") ?? null;
}

async function show(browser: BrowserState) {
  drawer = mount(JobsDrawer, { target: document.body, props: { browser } });
  flushSync();
  const bar = document.querySelector<HTMLButtonElement>(".bar");
  expect(bar).not.toBeNull();
  bar!.click();
  flushSync();
  await vi.waitFor(() => expect(browser.recoveryStatus).not.toBeNull());
}

function changeBudget(value: string): HTMLSelectElement {
  const select = budgetSelect();
  expect(select).not.toBeNull();
  select!.value = value;
  select!.dispatchEvent(new Event("change", { bubbles: true }));
  return select!;
}

function observe(caseName: string, phase: string, f: ReturnType<typeof fixture>, provider: AppSettingsDTO) {
  const select = budgetSelect();
  const record = {
    case: caseName, phase, provider,
    cache: f.browser.appSettings,
    select: select ? { value: select.value, disabled: select.disabled, options: [...select.options].map((o) => o.value) } : null,
    getCalls: vi.mocked(f.api.appSettings).mock.calls,
    setCalls: vi.mocked(f.api.saveAppSettings).mock.calls,
    text: document.body.textContent,
    html: document.body.innerHTML,
  };
  console.log("BUDGET_PROBE_OBSERVATION", JSON.stringify(record));
  return record;
}

it("control: acknowledged budget save preserves the complete document and awaits the reply", async () => {
  const f = fixture();
  const reply = deferred<AppSettingsDTO>({ ...initial, maxActiveJobs: 4 });
  f.api.saveAppSettings = vi.fn(() => reply.promise);
  await show(f.browser);
  await vi.waitFor(() => expect(budgetSelect()?.value).toBe("2"));
  const select = changeBudget("4");
  flushSync();
  expect(select.disabled).toBe(true);
  expect(f.browser.appSettings).toEqual(initial);
  expect(f.api.saveAppSettings).toHaveBeenCalledExactlyOnceWith({ ...initial, maxActiveJobs: 4 });
  observe("acknowledged", "pending", f, initial);
  reply.resolve({ ...initial, maxActiveJobs: 4 });
  await vi.waitFor(() => expect(select.disabled).toBe(false));
  expect(f.browser.appSettings).toEqual({ ...initial, maxActiveJobs: 4 });
  expect(select.value).toBe("4");
  expect([...select.options].map((option) => option.value)).toEqual(["1", "2", "4", "8"]);
  expect(f.api.appSettings).toHaveBeenCalledTimes(1);
  observe("acknowledged", "returned", f, { ...initial, maxActiveJobs: 4 });
});

it("control: an explicit state Get recovers a failed initial read without a Set", async () => {
  const f = fixture();
  f.api.appSettings = vi.fn().mockRejectedValueOnce(new Error("owned initial Get failure"))
    .mockResolvedValueOnce({ ...initial, maxActiveJobs: 8 });
  await show(f.browser);
  expect(f.browser.appSettings).toBeNull();
  observe("state_read_recovery", "initial_unavailable", f, { ...initial, maxActiveJobs: 8 });
  // This directly exercises the existing read method; no retry UI exists yet.
  await f.browser.loadAppSettings();
  flushSync();
  expect(f.browser.appSettings).toEqual({ ...initial, maxActiveJobs: 8 });
  expect(budgetSelect()?.value).toBe("8");
  expect(f.api.appSettings).toHaveBeenCalledTimes(2);
  expect(f.api.saveAppSettings).not.toHaveBeenCalled();
  observe("state_read_recovery", "read_returned", f, { ...initial, maxActiveJobs: 8 });
});

it("before: initial Get failure must remain visible with a read-only recovery control", async () => {
  const f = fixture();
  f.api.appSettings = vi.fn(async () => { throw new Error("owned Get failure"); });
  await show(f.browser);
  expect(f.browser.appSettings).toBeNull();
  expect(f.api.appSettings).toHaveBeenCalledTimes(1);
  expect(f.api.saveAppSettings).not.toHaveBeenCalled();
  const actual = observe("initial_get_failure", "returned", f, initial);
  expect.soft(actual.text, "UI10-BUDGET-GET: unavailable settings are not an absent control")
    .toMatch(/(?:budget|settings)[^.\n]*(?:unavailable|could not|unconfirmed)/i);
  const recovery = [...document.querySelectorAll("button")].find((button) =>
    /(?:retry|refresh).*(?:settings|budget)|(?:settings|budget).*(?:retry|refresh)/i.test(button.textContent ?? ""));
  expect(recovery, "UI10-BUDGET-RETRY: offer a read-only retry after Get failure").toBeDefined();
});

it("before: a failed later Get must qualify the retained budget as stale", async () => {
  const f = fixture();
  f.api.appSettings = vi.fn().mockResolvedValueOnce({ ...initial })
    .mockRejectedValueOnce(new Error("owned refresh Get failure"));
  await show(f.browser);
  await vi.waitFor(() => expect(budgetSelect()?.value).toBe("2"));
  await f.browser.loadAppSettings();
  flushSync();
  expect(f.browser.appSettings).toEqual(initial);
  expect(f.api.saveAppSettings).not.toHaveBeenCalled();
  const actual = observe("cached_get_failure", "returned", f, { ...initial, maxActiveJobs: 8 });
  expect(actual.text, "UI10-BUDGET-STALE: cached settings are not a successful current read")
    .toMatch(/(?:budget|settings)[^.\n]*(?:unavailable|could not|unconfirmed|stale|last (?:known|loaded))/i);
});

it.each([false, true])("before: rejected Set discloses an unconfirmed outcome (applied %s)", async (applied) => {
  const f = fixture();
  let provider = { ...initial };
  const reply = deferred<AppSettingsDTO>({ ...initial, maxActiveJobs: 4 });
  f.api.appSettings = vi.fn(async () => ({ ...provider }));
  f.api.saveAppSettings = vi.fn((settings) => {
    if (applied) provider = { ...settings };
    return reply.promise;
  });
  await show(f.browser);
  await vi.waitFor(() => expect(budgetSelect()?.value).toBe("2"));
  const select = changeBudget("4");
  flushSync();
  expect(select.disabled).toBe(true);
  reply.reject(new Error(applied ? "owned applied Set reply lost" : "owned Set refused before effect"));
  // The real void event callback must retain its unhandled rejection in Vitest.
  await vi.waitFor(() => expect(select.disabled).toBe(false));
  expect(f.browser.appSettings).toEqual(initial);
  expect(f.api.saveAppSettings).toHaveBeenCalledExactlyOnceWith({ ...initial, maxActiveJobs: 4 });
  expect(f.api.appSettings).toHaveBeenCalledTimes(1);
  const actual = observe(`set_rejected_applied_${applied}`, "rejected", f, provider);
  await f.browser.loadAppSettings();
  flushSync();
  observe(`set_rejected_applied_${applied}`, "explicit_state_get_returned", f, provider);
  expect(f.browser.appSettings).toEqual(provider);
  expect.soft(budgetSelect()?.value, "UI10-BUDGET-OBSERVED: a successful read must be reflected by the select")
    .toBe(String(provider.maxActiveJobs));
  expect(f.api.appSettings).toHaveBeenCalledTimes(2);
  expect(f.api.saveAppSettings).toHaveBeenCalledTimes(1);
  expect(actual.text, "UI10-BUDGET-SET: attempted selection is not acknowledgement")
    .toMatch(/(?:budget|settings|save)[^.\n]*(?:unconfirmed|could not be confirmed)/i);
});

it("before: synchronous duplicate DOM changes cannot admit a second pending Set", async () => {
  const f = fixture();
  const first = deferred<AppSettingsDTO>({ ...initial, maxActiveJobs: 4 });
  const second = deferred<AppSettingsDTO>({ ...initial, maxActiveJobs: 8 });
  f.api.saveAppSettings = vi.fn().mockReturnValueOnce(first.promise).mockReturnValueOnce(second.promise);
  await show(f.browser);
  await vi.waitFor(() => expect(budgetSelect()?.value).toBe("2"));
  // This is a same-turn synthetic DOM dispatch, not physical-user timing proof.
  const select = changeBudget("4");
  changeBudget("8");
  flushSync();
  const admitted = vi.mocked(f.api.saveAppSettings).mock.calls.length;
  expect(select.disabled).toBe(true);
  expect(f.browser.appSettings).toEqual(initial);
  observe("same_turn_duplicate", "both_dispatched", f, initial);
  first.resolve({ ...initial, maxActiveJobs: 4 });
  await vi.waitFor(() => expect(f.browser.appSettings?.maxActiveJobs).toBe(4));
  observe("same_turn_duplicate", "first_ack_second_pending", f, { ...initial, maxActiveJobs: 4 });
  second.resolve({ ...initial, maxActiveJobs: 8 });
  if (admitted > 1) await vi.waitFor(() => expect(f.browser.appSettings?.maxActiveJobs).toBe(8));
  flushSync();
  observe("same_turn_duplicate", "all_replies_released", f, f.browser.appSettings!);
  expect(admitted, "UI10-BUDGET-DUPLICATE: busy admission must precede the DOM render").toBe(1);
});
