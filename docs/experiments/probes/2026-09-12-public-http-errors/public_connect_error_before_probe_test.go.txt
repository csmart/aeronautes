package app

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"net/http/httptest"
	"path/filepath"
	"reflect"
	"strings"
	"sync"
	"testing"
	"time"

	"github.com/csmart/aeronautes/internal/bookmarks"
	"github.com/csmart/aeronautes/internal/db"
	"github.com/csmart/aeronautes/internal/evidence"
	"github.com/csmart/aeronautes/internal/secrets"
	"github.com/csmart/aeronautes/internal/swiftcontrol"
	"github.com/csmart/aeronautes/internal/swifttest"
	"github.com/ncw/swift/v2"
)

type connectErrorProbeRequest struct {
	Method         string `json:"method"`
	Path           string `json:"path"`
	Protocol       string `json:"protocol"`
	AuthKeyMatched bool   `json:"auth_key_matched"`
	ReasonSent     string `json:"reason_sent,omitempty"`
	BodySent       string `json:"body_sent,omitempty"`
	FixtureIOError string `json:"fixture_io_error,omitempty"`
}

func TestPublicConnectErrorBoundaryProbe(t *testing.T) {
	cases := []struct {
		name         string
		status       int
		customReason bool
	}{
		{"healthy", http.StatusOK, false},
		{"mapped_403", http.StatusForbidden, true},
		{"unmapped_standard_500", http.StatusInternalServerError, false},
		{"unmapped_custom_500", http.StatusInternalServerError, true},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			// These are deliberately synthetic; no personal store or credential is read.
			const user = "probe:public-error"
			const canary = "SYNTHETIC_CONNECT_KEY_b39126"
			const bodyCanary = "SYNTHETIC_BODY_ONLY_c057da"
			ctx, cancel := context.WithTimeout(t.Context(), 10*time.Second)
			defer cancel()
			d, err := db.Open(ctx, filepath.Join(t.TempDir(), "probe.sqlite"))
			if err != nil {
				t.Fatal(err)
			}
			var closeDBOnce sync.Once
			closeDB := func() {
				closeDBOnce.Do(func() {
					if err := d.Close(); err != nil {
						t.Errorf("fixture database close: %v", err)
					}
				})
			}
			t.Cleanup(closeDB)
			mock := swifttest.New(user, canary)
			var mu sync.Mutex
			var requests []connectErrorProbeRequest
			var handlers sync.WaitGroup
			server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				handlers.Add(1)
				defer handlers.Done()
				observed := connectErrorProbeRequest{Method: r.Method, Path: r.URL.Path, Protocol: r.Proto}
				defer func() {
					mu.Lock()
					requests = append(requests, observed)
					mu.Unlock()
				}()
				if r.URL.Path != "/auth/v1.0" {
					mock.ServeHTTP(w, r)
					return
				}
				observed.AuthKeyMatched = r.Header.Get("X-Auth-Key") == canary && r.Header.Get("X-Auth-User") == user
				if tc.status == http.StatusOK {
					mock.ServeHTTP(w, r)
					return
				}
				observed.ReasonSent = http.StatusText(tc.status)
				if tc.customReason {
					observed.ReasonSent = "reason-only-" + r.Header.Get("X-Auth-Key")
				}
				observed.BodySent = "body-only-" + bodyCanary
				// Hijacking preserves a provider-chosen reason; WriteHeader would replace it.
				conn, rw, err := w.(http.Hijacker).Hijack()
				if err != nil {
					observed.FixtureIOError = "hijack: " + err.Error()
					return
				}
				var ioErr error
				if err := conn.SetWriteDeadline(time.Now().Add(3 * time.Second)); err != nil {
					ioErr = err
				} else {
					_, ioErr = fmt.Fprintf(rw, "HTTP/1.1 %d %s\r\nContent-Type: text/plain\r\nContent-Length: %d\r\nConnection: close\r\n\r\n%s",
						tc.status, observed.ReasonSent, len(observed.BodySent), observed.BodySent)
					if ioErr == nil {
						ioErr = rw.Flush()
					}
				}
				if joined := errors.Join(ioErr, conn.Close()); joined != nil {
					observed.FixtureIOError = joined.Error()
				}
			}))
			var closeServerOnce sync.Once
			closeServer := func() {
				closeServerOnce.Do(func() {
					server.Close()
					// Server.Close does not join hijacked connections; this fixture owns them.
					handlers.Wait()
				})
			}
			t.Cleanup(closeServer)
			store := bookmarks.NewStore(d, secrets.NewSessionStore())
			connections := NewConnections(store)
			bookmarkService := NewBookmarkService(store, connections, nil)
			browse := NewBrowseService(connections, evidence.NewStore(d))
			saved, err := bookmarkService.Save(ctx, BookmarkDTO{
				DisplayName: "public-error-probe", AuthURL: server.URL + "/auth/v1.0", Username: user, TLSMode: "system",
			}, canary)
			if err != nil {
				t.Fatal(err)
			}
			before, err := bookmarkService.List(ctx)
			if err != nil {
				t.Fatal(err)
			}
			account, connectErr := browse.Connect(ctx, saved.ID)
			closeServer()
			after, err := bookmarkService.List(ctx)
			if err != nil {
				t.Fatal(err)
			}
			if !reflect.DeepEqual(before, after) {
				t.Error("Connect changed the saved bookmark")
			}
			bookmarkJSON, err := json.Marshal(after)
			if err != nil {
				t.Fatal(err)
			}
			if strings.Contains(string(bookmarkJSON), canary) {
				t.Error("bookmark DTO contains the synthetic key")
			}
			mu.Lock()
			observedRequests := append([]connectErrorProbeRequest(nil), requests...)
			mu.Unlock()
			authCount := 0
			for _, request := range observedRequests {
				if request.Protocol != "HTTP/1.1" || request.FixtureIOError != "" {
					t.Errorf("fixture protocol/I/O failure: %+v", request)
				}
				if request.Method != http.MethodGet && request.Method != http.MethodHead {
					t.Errorf("Connect issued a mutation: %+v", request)
				}
				if request.Path == "/auth/v1.0" {
					authCount++
					if !request.AuthKeyMatched {
						t.Error("fixture did not receive the saved synthetic credential")
					}
				}
			}
			if authCount != 1 {
				t.Errorf("auth requests = %d, want 1", authCount)
			}
			publicError, normalizedMessage, wireText, privateCause := "", "", "", ""
			kind, httpStatus := swiftcontrol.ErrorKind(""), 0
			if tc.status == http.StatusOK {
				if connectErr != nil || account.Name != "AUTH_test" || len(observedRequests) != 3 {
					t.Errorf("healthy public Connect: account=%+v error=%v requests=%d", account, connectErr, len(observedRequests))
				}
			} else if connectErr == nil {
				t.Error("failed authentication became a successful public Connect")
			} else {
				publicError = connectErr.Error()
				var se *swiftcontrol.SwiftError
				if !errors.As(connectErr, &se) {
					t.Errorf("public error type = %T, want SwiftError", connectErr)
				} else {
					normalizedMessage, kind, httpStatus = se.Message, se.Kind, se.HTTPStatus
					if se.HTTPStatus != tc.status {
						t.Errorf("public status = %d, want %d", se.HTTPStatus, tc.status)
					}
				}
				var we *swift.Error
				if !errors.As(connectErr, &we) {
					t.Error("native Swift error identity was lost")
				} else {
					wireText = we.Text
				}
				if unwrapped := errors.Unwrap(connectErr); unwrapped != nil {
					privateCause = unwrapped.Error()
				}
				if !reflect.DeepEqual(account, AccountDTO{}) || len(observedRequests) != 1 {
					t.Errorf("refused Connect returned data or reached storage: account=%+v requests=%+v", account, observedRequests)
				}
				if _, err := connections.Session(); !errors.Is(err, ErrNotConnected) {
					t.Errorf("failed authentication installed active authority: %v", err)
				}
			}
			closeDB()
			observation, err := json.Marshal(map[string]any{
				"case": tc.name, "requests": observedRequests, "public_error": publicError,
				"normalized_message": normalizedMessage, "native_error_text": wireText,
				"private_unwrapped_cause": privateCause, "kind": kind, "http_status": httpStatus,
				"account": account, "bookmarks_unchanged": reflect.DeepEqual(before, after),
				"server_closed_handlers_joined": true, "database_closed": true,
			})
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("PUBLIC_ERROR_OBSERVATION %s", observation)
			if strings.Contains(publicError, bodyCanary) {
				t.Error("SEC-PUBLIC-BODY: public Connect returned the response body canary")
			}
			if strings.Contains(publicError, canary) || strings.Contains(normalizedMessage, canary) {
				t.Error("SEC-PUBLIC-REASON: public Connect returned the synthetic saved key from the HTTP reason phrase")
			}
		})
	}
}
