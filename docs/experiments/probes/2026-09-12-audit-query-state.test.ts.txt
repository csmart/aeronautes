import { afterEach, expect, it, vi } from "vitest";
import { flushSync, mount, unmount } from "svelte";
import type { AuditEventDTO } from "../../lib/api";
import { BrowserState } from "../browser/browserState.svelte";
import { newFakeApi } from "../browser/fakeApi";
import JobsDrawer from "./JobsDrawer.svelte";

let drawer: ReturnType<typeof mount> | null = null;

afterEach(async () => {
  if (drawer) await unmount(drawer);
  drawer = null;
  document.body.innerHTML = "";
});

function audit(id: number, operation = "copy.object"): AuditEventDTO {
  return { id, at: "", operation, resource: `object-${id}`, summary: operation,
    result: "applied", transIds: ["tx-owned"] };
}

function deferred<T>() {
  let resolve!: (value: T) => void;
  const promise = new Promise<T>((done) => { resolve = done; });
  return { promise, resolve };
}

it("control: a successful empty search preserves literal all-account query values", async () => {
  const api = newFakeApi();
  api.auditSearch = vi.fn(async () => []);
  const browser = new BrowserState(api);
  browser.auditFilterOp = " delete.λ ";
  browser.auditFilterResult = "failed";
  await browser.searchAudit();
  expect(browser.auditEvents).toEqual([]);
  expect(browser.auditExhausted).toBe(true);
  expect(api.auditSearch).toHaveBeenCalledExactlyOnceWith({
    operationPrefix: " delete.λ ", result: "failed", bookmarkId: "", beforeId: 0, limit: 50,
  });
});

it("admits one More request after a legal full first page and appends each ID once", async () => {
  const api = newFakeApi();
  const page = deferred<AuditEventDTO[]>();
  const initial = Array.from({ length: 50 }, (_, index) => audit(100 - index));
  api.auditSearch = vi.fn().mockResolvedValueOnce(initial).mockReturnValue(page.promise);
  const browser = new BrowserState(api);
  await browser.searchAudit();
  expect(browser.auditEvents.map((event) => event.id)).toEqual(initial.map((event) => event.id));
  expect(browser.auditExhausted).toBe(false);
  const first = browser.searchAudit(true);
  const second = browser.searchAudit(true);
  page.resolve([audit(50), audit(49)]);
  await Promise.all([first, second]);
  console.info("AUDIT_LEGAL_MORE", JSON.stringify({
    calls: vi.mocked(api.auditSearch).mock.calls,
    ids: browser.auditEvents.map((event) => event.id),
  }));
  expect.soft(api.auditSearch).toHaveBeenCalledTimes(2);
  expect.soft(api.auditSearch).toHaveBeenNthCalledWith(2, {
    operationPrefix: "", result: "", bookmarkId: "", beforeId: 51, limit: 50,
  });
  expect(browser.auditEvents.map((event) => event.id)).toEqual([
    ...initial.map((event) => event.id), 50, 49,
  ]);
});

it("keeps the newer selected query when the old first page finishes last", async () => {
  const api = newFakeApi();
  const old = deferred<AuditEventDTO[]>();
  api.auditSearch = vi.fn().mockReturnValueOnce(old.promise)
    .mockResolvedValueOnce([audit(22, "delete.container")]);
  const browser = new BrowserState(api);
  browser.auditFilterOp = "copy.";
  const first = browser.searchAudit();
  browser.auditFilterOp = "delete.";
  await browser.searchAudit();
  old.resolve([audit(11)]);
  await first;
  expect(browser.auditEvents.map((event) => event.id)).toEqual([22]);
});

it("retires a pending query when filters change away and back without another search", async () => {
  const api = newFakeApi();
  const old = deferred<AuditEventDTO[]>();
  api.auditSearch = vi.fn(() => old.promise);
  const browser = new BrowserState(api);
  browser.auditFilterOp = "copy.";
  const first = browser.searchAudit();
  browser.auditFilterOp = "delete.";
  browser.auditFilterOp = "copy.";
  old.resolve([audit(11)]);
  await first;
  expect(api.auditSearch).toHaveBeenCalledTimes(1);
  expect(browser.auditEvents).toEqual([]);
});

it("keeps a late Recent refresh out of the selected filtered view", async () => {
  const api = newFakeApi();
  const recent = deferred<AuditEventDTO[]>();
  api.auditRecent = vi.fn(() => recent.promise);
  api.auditSearch = vi.fn(async () => [audit(22, "delete.container")]);
  const browser = new BrowserState(api);
  const refresh = browser.loadAudit();
  browser.auditFilterOp = "delete.";
  await browser.searchAudit();
  recent.resolve([audit(30)]);
  await refresh;
  expect(browser.auditEvents.map((event) => event.id)).toEqual([22]);
});

it("does not announce an empty audit result while the initial read is pending", async () => {
  const api = newFakeApi();
  const pending = deferred<AuditEventDTO[]>();
  api.auditSearch = vi.fn(() => pending.promise);
  const browser = new BrowserState(api);
  drawer = mount(JobsDrawer, { target: document.body, props: { browser } });
  flushSync();
  document.querySelector<HTMLButtonElement>(".bar")!.click();
  flushSync();
  const tab = [...document.querySelectorAll<HTMLButtonElement>("button")]
    .find((button) => button.textContent?.trim() === "Recent changes");
  expect(tab).toBeDefined();
  try {
    tab!.click();
    flushSync();
    expect(api.auditSearch).toHaveBeenCalledTimes(1);
    expect(document.body.textContent).not.toContain("No matching audit events");
  } finally {
    pending.resolve([]);
    await pending.promise;
    flushSync();
  }
});
