// Private UI fixture: current swifttest data, with all HTTP mutations refused.
package main

import (
	"encoding/json"
	"flag"
	"log"
	"net/http"
	"os"

	"github.com/csmart/aeronautes/internal/swifttest"
)

func main() {
	addr := flag.String("addr", "127.0.0.1:0", "owned loopback address")
	fixture := flag.String("fixture", "", "private fixture JSON")
	refusal := flag.String("refusal", "", "body returned with every refused method")
	flag.Parse()
	data, err := os.ReadFile(*fixture)
	if err != nil {
		log.Fatal(err)
	}
	var spec struct {
		Containers []string
		Objects    []struct {
			Container, Name string
			Object          swifttest.Object
		}
	}
	if err := json.Unmarshal(data, &spec); err != nil {
		log.Fatal(err)
	}
	s := swifttest.New("test:tester", "testing")
	for _, name := range spec.Containers {
		s.AddContainer(name, "standard")
	}
	for _, entry := range spec.Objects {
		s.PutObject(entry.Container, entry.Name, entry.Object)
	}
	log.Printf("seeded %d containers and %d objects from private fixture", len(spec.Containers), len(spec.Objects))
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		log.Printf("%s %s", r.Method, r.URL.RequestURI())
		if r.Method != http.MethodGet && r.Method != http.MethodHead {
			// A refusal before the data handler, with an optional long
			// unbroken token so the dialog's error path can be observed.
			w.WriteHeader(http.StatusMethodNotAllowed)
			if *refusal != "" {
				_, _ = w.Write([]byte(*refusal))
			}
			return
		}
		s.ServeHTTP(w, r)
	})
	log.Fatal(http.ListenAndServe(*addr, h))
}
