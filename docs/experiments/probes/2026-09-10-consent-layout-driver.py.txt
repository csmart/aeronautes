A sequence may declare allow_activation, which permits activating a control
that cannot destroy data, and expect_refused_writes, which replaces the
no-write invariant with a refused-attempt invariant. The private fixture
answers every non-read request with 405 before touching data, and the
complete remote snapshot is compared either way."""

def request(path, method='GET', data=None, token=None, extra=None):
    headers = {'X-Auth-Token': token} if token else {'X-Auth-User': 'test:tester', 'X-Auth-Key': 'testing'}
    headers.update(extra or {})
    with urllib.request.urlopen(urllib.request.Request(origin + path, data=data, method=method, headers=headers), timeout=5) as reply:
        return reply.headers, reply.read()


def wire(pattern, seconds):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        if re.search(pattern, (out / 'mockswift.log').read_text()[wire_start:], re.MULTILINE):
            return True
        time.sleep(.2)
    return False

def read_window_rect():
    shell = x('getwindowgeometry', '--shell', window).stdout
    values = dict(line.split('=', 1) for line in shell.strip().splitlines() if '=' in line)
    left, top = int(values['X']), int(values['Y'])
    return (left, top, left + int(values['WIDTH']), top + int(values['HEIGHT']))


def inspect(name):
    """Capture and measure the dialog without disturbing the run."""
    path = out / (name + '.png')
    capture(env['DISPLAY'], path)
    rows = geometry.pixels(path)
    box = geometry.dialog(rows, window=window_rect)
    record = {'screenshot': name, 'window_rect': window_rect, 'dialog': box}
    if box:
        left, top, right, bottom = window_rect
        record['fits_window'] = box['top'] > top and box['bottom'] < bottom - 1
        record['clipped_top'] = box['top'] <= top
        record['clipped_bottom'] = box['bottom'] >= bottom - 1
        record['checkbox_hits'] = geometry.find_patch(rows, TEMPLATES[box['theme']], box)
    return record, rows, box

            result['app_http_mutations'] = [m for m in methods if m not in ('GET', 'HEAD')]
            after = snapshot()
            (out / 'remote-after-finally.json').write_text(json.dumps(after, indent=2) + '\n')
            result['final_snapshot_equal'] = after == baseline
            unexpected = [m for m in result['app_http_mutations'] if not (expect_refused_writes and m == 'DELETE')]
            if unexpected or not result['final_snapshot_equal']:
                raise AssertionError('final data or write invariant failed')
    except BaseException as error:
        result['final_snapshot_error'] = str(error)
        result['status'] = 'failed'
    result['finished_utc'] = time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
    save()
    for process in reversed(processes):
        if process.poll() is None:
            process.send_signal(signal.SIGCONT)
            process.terminate()
            try:
                process.wait(timeout=5)
            except subprocess.TimeoutExpired:
                process.kill(); process.wait(timeout=5)
