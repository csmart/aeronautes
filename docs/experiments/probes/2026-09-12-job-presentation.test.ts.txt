import { afterEach, expect, it, vi } from "vitest";
import { flushSync, mount, unmount } from "svelte";
import type { JobDTO } from "../../lib/api";
import { BrowserState } from "../browser/browserState.svelte";
import { newFakeApi } from "../browser/fakeApi";
import JobsDrawer from "./JobsDrawer.svelte";

let drawer: ReturnType<typeof mount> | null = null;

afterEach(async () => {
  if (drawer) await unmount(drawer);
  drawer = null;
  document.body.innerHTML = "";
  vi.restoreAllMocks();
});

function job(type: string, state: string, counters: Partial<JobDTO["counters"]> = {}): JobDTO {
  return {
    id: `job-${type}`, type, state,
    source: "source/scope/", destination: "target/scope/",
    sourceBookmarkId: "a", destBookmarkId: "b",
    createdAt: "2026-09-12T00:00:00Z", startedAt: "2026-09-12T00:00:01Z",
    finishedAt: state === "partial" ? "2026-09-12T00:00:09Z" : "",
    engine: "rclone+native", settings: '{"transfers":2,"checkers":4}',
    error: "", recoveryRecords: 0,
    counters: {
      objectsPlanned: 10, objectsCopied: 3, objectsVerified: 0, objectsFailed: 0,
      bytesPlanned: 80, bytesCopied: 24, overwrites: 0, overwritesRefused: 0,
      nativeRouted: 0, objectsDeleted: 0, ...counters,
    },
  };
}

function deletion(type: string, state: string): JobDTO {
  const value = job(type, state, {
    bytesCopied: 0,
    objectsFailed: state === "partial" ? 1 : 0,
    objectsVerified: state === "partial" ? 3 : 0,
  });
  // The real DeleteService uses objectsCopied for reported deletions;
  // objectsDeleted is reserved for the historical Mirror executor.
  return { ...value, destination: "(deleted)", destBookmarkId: "", engine: "native", settings: "" };
}

async function show(value: JobDTO) {
  const api = newFakeApi();
  const original = JSON.stringify(value);
  api.jobs = vi.fn(async () => [structuredClone(value)]);
  api.job = vi.fn(async () => structuredClone(value));
  api.cancelJob = vi.fn(async () => { throw new Error("unexpected mutation"); });
  api.rerunJob = vi.fn(async () => { throw new Error("unexpected mutation"); });
  const browser = new BrowserState(api);
  drawer = mount(JobsDrawer, { target: document.body, props: { browser } });
  flushSync();
  document.querySelector<HTMLButtonElement>(".bar")!.click();
  flushSync();
  await vi.waitFor(() => expect(document.querySelectorAll(".job")).toHaveLength(1));
  expect(api.jobs).toHaveBeenCalledWith("", false, 50);
  expect(JSON.stringify(value)).toBe(original);
  const row = document.querySelector<HTMLElement>(".job")!;
  return { api, browser, row };
}

function observe(name: string, value: JobDTO, row: HTMLElement, browser: BrowserState) {
  console.info("JOB_PRESENTATION_OBSERVATION " + JSON.stringify({
    name, dto: value,
    row: row.textContent?.replace(/\s+/g, " ").trim(),
    counts: row.querySelector(".counts")?.textContent?.replace(/\s+/g, " ").trim(),
    config: row.querySelector(".config")?.textContent?.replace(/\s+/g, " ").trim() ?? null,
    progress: row.querySelector(".progress")?.getAttribute("style") ?? null,
    rate: browser.jobRates[value.id] ?? null,
    actions: [...row.querySelectorAll(".row-actions button")].map((button) => ({
      text: button.textContent?.trim(), label: button.getAttribute("aria-label"),
    })),
  }));
}

const deletionCases = [
  { type: "delete_prefix", state: "running", label: "Delete prefix" },
  { type: "empty_container", state: "partial", label: "Empty container" },
  { type: "delete_container", state: "partial", label: "Delete container" },
];

it.each(deletionCases)("uses a deletion noun without transfer bytes for $type $state", async ({ type, state }) => {
  const value = deletion(type, state);
  const { browser, row } = await show(value);
  observe("deletion noun", value, row, browser);
  const counts = row.querySelector(".counts")?.textContent;
  expect(counts).toMatch(/3\s*\/10\s+objects reported deleted/);
  expect(counts).not.toContain("0 B");
});

it.each([
  ...deletionCases,
  { type: "copy", state: "running", label: "Copy" },
  { type: "upload", state: "running", label: "Upload" },
  { type: "download", state: "partial", label: "Download" },
])("visibly names the actual $type operation in its row", async ({ type, state, label }) => {
  const value = deletionCases.some((entry) => entry.type === type) ? deletion(type, state) : job(type, state);
  const { browser, row } = await show(value);
  observe("operation label", value, row, browser);
  expect(row.textContent).toContain(label);
});

it("does not call unrecorded running Delete settings defaults", async () => {
  const value = deletion("delete_prefix", "running");
  const { api, browser, row } = await show(value);
  row.querySelector<HTMLButtonElement>(`[aria-label="Item details for job ${value.id}"]`)!.click();
  flushSync();
  await vi.waitFor(() => expect(row.querySelector(".config")).not.toBeNull());
  observe("unrecorded settings", value, row, browser);
  expect(row.querySelector(".config")?.textContent).toContain("not recorded");
  expect(row.querySelector(".config")?.textContent).not.toContain("defaults");
  expect(api.cancelJob).not.toHaveBeenCalled();
  expect(api.rerunJob).not.toHaveBeenCalled();
});

it.each(deletionCases)("control: preserves actual Delete numerator and terminal qualification for $type", async ({ type, state }) => {
  const value = deletion(type, state);
  const { browser, row } = await show(value);
  observe("actual Delete numerator", value, row, browser);
  expect(row.querySelector(".counts")?.textContent).toMatch(/3\s*\/10/);
  expect(row.querySelector(".counts")?.textContent).not.toMatch(/0\s*\/10/);
  expect(value.counters.objectsDeleted).toBe(0);
  if (state === "running") {
    expect(row.querySelector(".progress")?.getAttribute("style")).toContain("30%");
    expect(row.querySelector<HTMLButtonElement>(".cancel")?.disabled).toBe(false);
  } else {
    expect(row.querySelector(".progress")).toBeNull();
    expect(row.querySelector(".counts")?.textContent).toContain("1 failed");
    expect(row.querySelector(".cancel")).toBeNull();
  }
});

it.each(["copy", "upload", "download"])("control: %s keeps counters and allows running diagnostics without cancellation", async (type) => {
  const value = job(type, "running");
  const { api, browser, row } = await show(value);
  const details = row.querySelector<HTMLButtonElement>(`[aria-label="Item details for job ${value.id}"]`)!;
  expect(details).not.toBeNull();
  details.click();
  flushSync();
  await vi.waitFor(() => expect(row.querySelector(".item-records")?.textContent).toContain("Job still active"));
  observe("live diagnostics already available", value, row, browser);
  expect(details.getAttribute("aria-expanded")).toBe("true");
  expect(row.querySelector(".config")?.textContent).toContain("rclone+native");
  expect(row.querySelector(".config")?.textContent).toContain("transfers2");
  expect(row.querySelector(".config")?.textContent).toContain("checkers4");
  expect(row.querySelector(".counts")?.textContent).toMatch(/3\s*\/10/);
  expect(row.querySelector(".counts")?.textContent).toContain("24 B");
  expect(row.querySelector(".progress")?.getAttribute("style")).toContain("30%");
  expect(api.cancelJob).not.toHaveBeenCalled();
  expect(api.rerunJob).not.toHaveBeenCalled();
});

it("control: incomplete deletion accounting remains last confirmed without final totals", async () => {
  const value = deletion("delete_container", "partial");
  value.counters.deletionAccountingIncomplete = true;
  value.error = "Retained admitted-attempt evidence is incomplete.";
  const { browser, row } = await show(value);
  row.querySelector<HTMLButtonElement>(`[aria-label="Item details for job ${value.id}"]`)!.click();
  flushSync();
  observe("incomplete deletion", value, row, browser);
  expect(row.querySelector(".counts")?.textContent).toContain("Last confirmed progress: 3 processed");
  expect(row.querySelector(".counts")?.textContent).toContain("1 reported failures");
  expect(row.textContent).toContain("Final failure and unresolved totals are unavailable");
  expect(row.querySelector(".counts")?.textContent).not.toMatch(/3\s*\/10/);
  expect(row.querySelector(".progress")).toBeNull();
  expect(row.querySelector(".config")?.textContent).not.toContain("verified");
  expect(row.querySelector(".error")?.textContent).toBe(value.error);
});

it("control: unknown planned count and first-poll rate do not invent a percent or ETA", async () => {
  const value = job("copy", "running", { objectsPlanned: 0, bytesPlanned: 0 });
  const { browser, row } = await show(value);
  observe("unknown telemetry", value, row, browser);
  expect(browser.jobRates[value.id]).toBeUndefined();
  expect(row.querySelector(".progress")).toBeNull();
  expect(row.querySelector(".when")?.textContent).toContain("2026-09-12 00:00:01");
  expect(row.querySelector(".when")?.textContent).not.toMatch(/NaN|Infinity|~|\/s/);
  expect(row.querySelector(".counts")?.textContent).not.toContain("/0");
});

it("control: historical Mirror keeps destination deletions separate from copied objects", async () => {
  const value = job("mirror", "partial", { objectsDeleted: 7, objectsFailed: 1 });
  const { browser, row } = await show(value);
  observe("historical Mirror", value, row, browser);
  expect(row.querySelector(".counts")?.textContent).toMatch(/3\s*\/10/);
  expect(row.querySelector(".counts")?.textContent).toContain("7 deleted");
  expect(row.querySelector(".counts")?.textContent).toContain("1 failed");
  expect(row.querySelector(".progress")).toBeNull();
  expect(row.querySelector(".rerun")).toBeNull();
});
