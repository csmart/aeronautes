type auditCorruptStageEngine struct{ inner *fakeEngine }

func (e auditCorruptStageEngine) Copy(ctx context.Context, src, dst string, names []string, onEvent func(rclone.Event)) (*rclone.Result, error) {
	r, err := e.inner.Copy(ctx, src, dst, names, onEvent)
	if err == nil && len(r.SegmentPrefixes) > 0 {
		p := r.SegmentPrefixes[0]
		seed(e.inner.w.dstMock, p.Container, p.Prefix+"00000000", []byte("EVIL"), nil)
	}
	return r, err
}

func (e auditCorruptStageEngine) CopyTo(ctx context.Context, src, dst string, onEvent func(rclone.Event)) (*rclone.Result, error) {
	return e.inner.CopyTo(ctx, src, dst, onEvent)
}

func TestAuditCopyRejectsSameSizeCorruptSegments(t *testing.T) {
	w := newWorld(t)
	ctx := context.Background()
	seed(w.srcMock, "media", "asset", []byte("GOODDATA"), nil)
	e := newExecutor(t, w, auditCorruptStageEngine{inner: &fakeEngine{w: w, chunk: 4, interruptAfter: -1}})
	r, err := e.Run(ctx, []plan.Item{{Name: "asset", Verdict: plan.VerdictNew, Route: plan.RouteRclone, SourceBytes: 8}})
	if err != nil {
		t.Fatal(err)
	}
	body, _, err := w.dst.OpenObject(ctx, "backup", "asset")
	if err != nil {
		t.Logf("live copy refused: %v", err)
		return
	}
	defer body.Close()
	var out bytes.Buffer
	if _, err := out.ReadFrom(body); err != nil {
		t.Fatal(err)
	}
	if out.String() != "GOODDATA" && r.Copied > 0 {
		t.Fatalf("data corruption reported copied: logical bytes=%q, report=%+v", out.String(), r)
	}
}

func TestAuditSkipRejectsChangedDestinationSegments(t *testing.T) {
	w := newWorld(t)
	ctx := context.Background()
	seed(w.srcMock, "media", "asset", []byte("GOODDATA"), nil)
	e := newExecutor(t, w, &fakeEngine{w: w, chunk: 4, interruptAfter: -1})
	r, err := e.Run(ctx, []plan.Item{{Name: "asset", Verdict: plan.VerdictNew, Route: plan.RouteRclone, SourceBytes: 8}})
	if err != nil || r.Copied != 1 {
		t.Fatalf("initial copy=%+v err=%v", r, err)
	}
	p := r.CapturedPrefixes[0]
	seed(w.dstMock, p.Container, p.Prefix+"00000000", []byte("EVIL"), nil)
	r2, err := newExecutor(t, w, &fakeEngine{w: w, chunk: 4, interruptAfter: -1}).Run(ctx,
		[]plan.Item{{Name: "asset", Verdict: plan.VerdictUnknown, Route: plan.RouteRclone, SourceBytes: 8}})
	if err != nil {
		t.Fatal(err)
	}
	if r2.SkippedEqual > 0 {
		t.Fatalf("changed logical destination is skipped as equal: %+v", r2)
	}
}

func TestAuditSweepKeepsExternallySharedCapturedPrefix(t *testing.T) {
	w := newWorld(t)
	ctx := context.Background()
	p := rclone.SegmentPrefix{Container: "backup_segments", Prefix: ".aeronautes/stage/old/batch/file/2026-random/"}
	seed(w.dstMock, p.Container, p.Prefix+"00000000", []byte("LIVEDATA"), nil)
	if _, err := w.dst.PutObject(ctx, "backup", "external-alias", strings.NewReader(""), "application/octet-stream",
		map[string]string{"X-Object-Manifest": p.Container + "/" + p.Prefix}); err != nil {
		t.Fatal(err)
	}
	r, err := SweepLitter(ctx, w.dst, SweepInput{DstContainer: "backup", Captured: []rclone.SegmentPrefix{p}})
	if err != nil {
		t.Fatal(err)
	}
	body, _, err := w.dst.OpenObject(ctx, "backup", "external-alias")
	if err != nil {
		t.Fatalf("live alias unreadable after sweep: %v; report=%+v", err, r)
	}
	defer body.Close()
	var out bytes.Buffer
	if _, err := out.ReadFrom(body); err != nil {
		t.Fatal(err)
	}
	if out.String() != "LIVEDATA" {
		t.Fatalf("sweep damaged an unrelated live alias: bytes=%q, report=%+v", out.String(), r)
	}
}

func TestAuditOverwritePreservesLastHopConcurrentWriter(t *testing.T) {
	w := newWorld(t)
	ctx := context.Background()
	seed(w.srcMock, "media", "asset", []byte("NEW"), nil)
	seed(w.dstMock, "backup", "asset", []byte("OLD"), nil)
	old := headDst(t, w, "asset")
	heads := 0
	w.dstMock.AfterRequest = func(method, path string) {
		if method == "HEAD" && path == "/v1/AUTH_test/backup/asset" {
			heads++
			if heads == 2 {
				seed(w.dstMock, "backup", "asset", []byte("RACER"), nil)
			}
		}
	}
	r, err := newExecutor(t, w, &fakeEngine{w: w, interruptAfter: -1}).Run(ctx,
		[]plan.Item{{Name: "asset", Verdict: plan.VerdictChanged, Route: plan.RouteRclone, DestETag: old.ETag, SourceBytes: 3}})
	w.dstMock.AfterRequest = nil
	if err != nil {
		t.Fatal(err)
	}
	if heads < 2 {
		t.Fatalf("probe did not hit final re-HEAD: %d", heads)
	}
	body, _, err := w.dst.OpenObject(ctx, "backup", "asset")
	if err != nil {
		t.Fatal(err)
	}
	defer body.Close()
	var out bytes.Buffer
	if _, err := out.ReadFrom(body); err != nil {
		t.Fatal(err)
	}
	if out.String() != "RACER" {
		t.Fatalf("concurrent writer lost at final commit: bytes=%q, report=%+v", out.String(), r)
	}
}
